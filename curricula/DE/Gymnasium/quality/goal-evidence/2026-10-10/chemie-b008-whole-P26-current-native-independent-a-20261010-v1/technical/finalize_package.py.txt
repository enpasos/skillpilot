import datetime
import hashlib
import importlib.util
import json
import pathlib
import shutil
import jsonschema

root = pathlib.Path.cwd()
out = root / 'curricula/DE/Gymnasium/quality/goal-evidence/2026-10-10/chemie-b008-whole-P26-current-native-independent-a-20261010-v1'
src = root / 'curricula/DE/Gymnasium/quality/goal-evidence/2026-10-10/chemie-b008-current-whole-P26-native-continuation-author-20261010-v1'

def now():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()

def digest(data):
    return 'sha256:' + hashlib.sha256(data).hexdigest()

def binding(path):
    data = path.read_bytes()
    return {'path': str(path.relative_to(root)), 'sha256': digest(data), 'bytes': len(data)}

def new_json(path, value):
    assert not path.exists(), 'Do not replace immutable review receipts'
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')

for name in ['seal_first.py', 'materialize_normal_records.py', 'check_p_capsule.py', 'finalize_package.py']:
    actual = root / 'tmp/chemie-P26-native-independent-a' / name
    target = out / 'technical' / (name + '.txt')
    assert not target.exists()
    target.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(actual, target)

first_path = out / 'FIRST.current-whole26-description-native.verdict.json'
first = json.loads(first_path.read_text())
assert digest(first_path.read_bytes()) == 'sha256:4aa2a7ad6d8cafccf7a18a030bb4852610c4e6bdd4b43807c87b40feed0d0172'
assert len(first['records']) == 26
assert all(row['decision'] == 'keep' for row in first['records'])
first_by_id = {row['goalId']: row for row in first['records']}
record_schema = json.loads((root / 'contracts/goal-description-review/v1/goal-description-review-record.schema.json').read_text())
run_schema = json.loads((root / 'contracts/goal-evidence/v1/goal-evidence-ai-run-manifest.schema.json').read_text())
record_validator = jsonschema.Draft202012Validator(record_schema, format_checker=jsonschema.FormatChecker())
run_validator = jsonschema.Draft202012Validator(run_schema, format_checker=jsonschema.FormatChecker())
json_count = 0
jsonl_count = 0
record_count = 0
errors = []
for path in sorted(out.rglob('*')):
    if path.is_symlink():
        errors.append(str(path) + ': review package symlink forbidden')
    if not path.is_file():
        continue
    if path.suffix == '.json':
        value = json.loads(path.read_bytes())
        json_count += 1
        if path.name.endswith('.run.json'):
            run_validator.validate(value)
    elif path.suffix == '.jsonl':
        data = path.read_bytes()
        assert data.endswith(b'\n')
        for line in data.decode().splitlines():
            assert line.strip()
            value = json.loads(line)
            record_validator.validate(value)
            reference = first_by_id[value['goalId']]
            assert value['decision'] == reference['decision']
            assert value['understandingEvidence'] == reference['understandingEvidence']
            assert value['goalFingerprint'] == reference['goalFingerprint']
            assert value['pageFingerprint'] == reference['pageFingerprint']
            record_count += 1
        jsonl_count += 1
assert record_count == 26

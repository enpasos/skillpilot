def bind(path):
    data = path.read_bytes()
    return {'path': str(path), 'sha256': 'sha256:' + hashlib.sha256(data).hexdigest(), 'bytes': len(data)}

def write(name, data):
    path = OUT / name
    assert not path.exists(), path
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(data, ensure_ascii=False, indent=2) + '\n')
    return path

def canonical_object_digest(data):
    return 'sha256:' + hashlib.sha256(json.dumps(data, ensure_ascii=False, sort_keys=True, separators=(',', ':')).encode()).hexdigest()

errors = []
for path in sorted(OUT.rglob('*')):
    assert not path.is_symlink(), path
    if not path.is_file():
        continue
    if path.suffix == '.json':
        value = json.loads(path.read_text())
        parsed_json += 1
        assert normal.validate_file(str(path), runtime_schema), path
        if path.name.endswith('.run.json'):
            validators['run'].validate(value)
            runs += 1
        if path.name.endswith('.normal.config.json'):
            validators['config'].validate(value)
            configs += 1
    elif path.suffix == '.jsonl':
        data = path.read_bytes()
        assert data.endswith(b'\n')
        parsed_jsonl += 1
        for line in data.decode().splitlines():
            assert line.strip()
            record = json.loads(line)
            if 'positive' in path.parts:
                validators['profile'].validate(record)
                assert record == material[record['goalId']]['wholeCurrentTechnicalCandidateRecord']
                p_records += 1
            else:
                validators['record'].validate(record)
                own = first_by_id[record['goalId']]
                assert record['decision'] == own['decision']
                assert record['goalFingerprint'] == own['goalFingerprint']
                assert record['pageFingerprint'] == own['pageFingerprint']
                evidence = {k: v for k, v in own['understandingEvidence'].items() if not k.startswith('transferCheck')}
                evidence['transferExpectationDe'] = own['understandingEvidence']['transferCheckDe']
                evidence['transferExpectationEn'] = own['understandingEvidence']['transferCheckEn']
                assert record['understandingEvidence'] == evidence
                description_records += 1

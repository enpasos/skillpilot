def score_work(d,kind,facet=None):
    sections=[];marks=[]
    for s,items in enumerate(d['criteria']):
        partsD=[];partsE=[];points=0;itemMarks=[]
        for n,it in enumerate(items):
            pd=it['answerDe'].split('; ',1);pe=it['answerEn'].split('; ',1)
            fs=it.get('partFacets',[it['facet'],it['facet']])
            assert len(pd)==len(pe)==len(fs)==2
            if kind=='full':present=[True,True]
            elif kind=='partial':present=[True,n==0]
            else:present=[f!=facet for f in fs]
            chosenD=[pd[j].rstrip('.')+'.' for j in range(2) if present[j]]
            chosenE=[pe[j].rstrip('.')+'.' for j in range(2) if present[j]]
            partsD.extend(chosenD);partsE.extend(chosenE)
            val=sum(present);points+=val;itemMarks.append({'criterion':n+1,'awarded':val,'max':2,'presentParts':present,'partFacets':fs,'rationale':'First/second independently stated correct component earns1; omitted component earns0. Author marking, not independent review.'})
        sections.append({'task':s+1,'answerDe':' '.join(partsD),'answerEn':' '.join(partsE),'marks':itemMarks,'points':points})
        marks.append(points)
    if kind=='absence':
        fd,fe=d['facets'][facet]
        sections.append({'declarationDe':'Die gesamte konkrete Leistung zu '+fd+' lasse ich in dieser Arbeit aus. Allgemeine Rechnungen oder Aussagen zu anderen Facetten sollen sie hier nicht ersetzen.','declarationEn':'I omit the entire specific performance concerning '+fe+' in this submission. General calculations or statements on other facets do not substitute for it.'})
    raw=sum(marks);final=min(raw,14) if kind=='absence' else raw
    return {'kind':kind,'omittedEssentialFacet':facet,'wholeAnswer':sections,'taskScores':marks,'rawPoints':raw,'finalPoints':final,'passed':final>=15,'authorDecisionOnly':True,'capReason':('Entire indicated essential facet absent across the complete submission, not merely missing one detail/task.' if kind=='absence' else None)}

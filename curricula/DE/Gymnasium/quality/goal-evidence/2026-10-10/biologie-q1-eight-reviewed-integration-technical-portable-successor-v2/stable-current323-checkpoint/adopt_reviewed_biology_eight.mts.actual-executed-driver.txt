const verify = (v: any): void => {
  if (Array.isArray(v)) return v.forEach(verify)
  if (!v || typeof v !== 'object') return
  if (typeof v.path === 'string' && typeof v.sha256 === 'string') {
    assert.ok(!v.path.startsWith('/') && !v.path.split('/').includes('..'))
    const actual = bind(v.path)
    assert.equal(actual.sha256,v.sha256.replace(/^sha256:/,''),v.path)
    if (v.bytes !== undefined) assert.equal(actual.bytes,v.bytes,v.path)
  }
  Object.values(v).forEach(verify)
}

}
